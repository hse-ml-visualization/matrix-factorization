1. SVD — сингулярное разложение

усечение ранга k
Что делает: раскладывает матрицу A на три сомножителя: A = U · Σ · Vᵀ. Усечение до ранга k оставляет только k наибольших сингулярных чисел — это даёт наилучшее приближение исходной матрицы. Где используется: сжатие изображений, шумоподавление, рекомендательные системы.
🧩 Пример из жизни: Представьте фотографию 1000×1000 пикселей. SVD с k=100 может сжать её в 10 раз, сохранив визуальное качество — вместо хранения всех 1 000 000 значений храним только 100 · (1000 + 1000) = 200 000.
JavaScript
// Усечённое SVD через numeric.js
export function svdTruncated(A, k) {
  // Вызываем полное SVD из библиотеки numeric
  const { U, S, V } = numeric.svd(A);
  const { m, n } = dims(A);
  // Реальный ранг — минимум из запрошенного k и доступных размеров
  const r = Math.max(1, Math.min(k, S.length,
    U[0].length, V.length, m, n));
  // Берём первые r столбцов U (левые сингулярные векторы)
  const Uk = takeUk(U, r);
  // Берём первые r сингулярных чисел
  const Sk = S.slice(0, r);
  // Берём первые r строк V (правые сингулярные векторы)
  const Vk = takeVk(V, r);
  // U·Σ — взвешенные левые векторы
  const US = dot(Uk, diag(Sk));
  // (U·Σ)·Vᵀ = Ã — приближение ранга r
  const Ahat = dot(US, transpose(Vk));
  return { Ahat, U, S, V, r, Uk, Sk, Vk, US };
}
Python
# Усечённое SVD через numpy.linalg.svd
import numpy as np

def svd_truncated(A, k):
    # Полное SVD: A = U·S·V†
    # full_matrices=False — экономный формат
    U, S, Vt = np.linalg.svd(A, full_matrices=False)
    # Реальный ранг — минимум из k и размера S
    r = max(1, min(k, len(S)))
    # Берём первые r столбцов U и строк V†
    Uk = U[:, :r]
    Sk = S[:r]
    Vkt = Vt[:r, :]
    # Σ как диагональная матрица r×r
    Sigma = np.diag(Sk)
    # U·Σ
    US = Uk @ Sigma
    # (U·Σ)·V† = Ã
    Ahat = US @ Vkt
    return {'Ahat': Ahat, 'r': r,
            'Uk': Uk, 'Sk': Sk, 'Vkt': Vkt}

2. PCA — метод главных компонент

центрирование + SVD
Что делает: центрирует данные (вычитает среднее по каждому столбцу), затем применяет SVD к центрированной матрице. Первые k главных компонент описывают наибольшую дисперсию данных. Где используется: уменьшение размерности, визуализация, сжатие признаков в ML.
🧩 Пример из жизни: У вас 50 характеристик товара (цена, вес, размер…). PCA может выделить 3 главные компоненты, которые объясняют 90% вариации — вы сокращаете данные в 16 раз, сохраняя основную структуру.
JavaScript
// PCA через центрирование + SVD
export function pcaReconstructWithSteps(A, k) {
  const { m, n } = dims(A);
  // Шаг 1: считаем среднее каждого столбца
  const mean = new Array(n).fill(0);
  for (let j = 0; j < n; j++) {
    for (let i = 0; i < m; i++) mean[j] += A[i][j];
    mean[j] /= m;
  }
  // Шаг 2: центрируем — вычитаем среднее
  const X = zeros(m, n);
  for (let i = 0; i < m; i++)
    for (let j = 0; j < n; j++) X[i][j] = A[i][j] - mean[j];
  // Шаг 3: SVD центрированной матрицы (усечённое)
  const svdX = svdTruncated(X, k);
  const { Ahat: Xk } = svdX;
  // Шаг 4: прибавляем среднее обратно — получаем Ã
  const Ahat = zeros(m, n);
  for (let i = 0; i < m; i++)
    for (let j = 0; j < n; j++) Ahat[i][j] = Xk[i][j] + mean[j];
  return { Ahat, mean, X, ...svdX };
}
Python
# PCA: центрирование + SVD (или sklearn)
import numpy as np
from sklearn.decomposition import PCA

def pca_reconstruct(A, k):
    # Шаг 1: среднее по столбцам
    mean = A.mean(axis=0)
    # Шаг 2: центрируем
    X = A - mean
    # Шаг 3: SVD центрированной матрицы
    U, S, Vt = np.linalg.svd(X, full_matrices=False)
    r = min(k, len(S))
    # Усечённое приближение X̂ = U·Σ·V†
    Xk = U[:, :r] @ np.diag(S[:r]) @ Vt[:r, :]
    # Шаг 4: прибавляем среднее обратно
    Ahat = Xk + mean
    return Ahat

# Или через sklearn (одна строка):
def pca_sklearn(A, k):
    pca = PCA(n_components=k)
    Z = pca.fit_transform(A)        # проекция на k компонент
    Ahat = pca.inverse_transform(Z)   # восстановление
    return Ahat

3. NMF — неотрицательное матричное разложение

мультипликативное обновление
Что делает: ищет две неотрицательные матрицы W и H такие, что A ≈ W·H. Обновление по правилу: H ← H · (WᵀA) / (WᵀWH), W ← W · (AHᵀ) / (WHHᵀ). Где используется: тематическое моделирование текстов, разложение спектров, разделение аудио, рекомендации.
🧩 Пример из жизни: Есть 1000 документов и 5000 уникальных слов. NMF с k=10 находит 10 «тем»: W говорит, сколько каждая тема есть в каждом документе, H — какие слова образуют каждую тему. Человек читает 10 списков слов вместо 1000 текстов.
JavaScript
// NMF — мультипликативное обновление (MU)
export function nmfReconstructHistory(A, k, totalIters) {
  const { m, n } = dims(A);
  // Если есть отрицательные — сдвигаем вверх
  const shift = minMax(A).mn < 0 ? -minMax(A).mn : 0;
  const X = A.map(r => r.map(v => v + shift));

  // Случайная инициализация W (m×k) и H (k×n)
  let W = zeros(m, k), H = zeros(k, n);
  for (let i = 0; i < m; i++)
    for (let t = 0; t < k; t++) W[i][t] = rnd();
  for (let t = 0; t < k; t++)
    for (let j = 0; j < n; j++) H[t][j] = rnd();

  for (let it = 0; it < totalIters; it++) {
    // Обновление H: H ← H · (WᵀX) / (WᵀW·H)
    const WT_X = dot(transpose(W), X);
    const WT_W_H = dot(dot(transpose(W), W), H);
    for (let t = 0; t < k; t++)
      for (let j = 0; j < n; j++)
        H[t][j] *= WT_X[t][j] / (WT_W_H[t][j] + 1e-9);

    // Обновление W: W ← W · (X·Hᵀ) / (W·H·Hᵀ)
    const X_HT = dot(X, transpose(H));
    const W_H_HT = dot(W, dot(H, transpose(H)));
    for (let i = 0; i < m; i++)
      for (let t = 0; t < k; t++)
        W[i][t] *= X_HT[i][t] / (W_H_HT[i][t] + 1e-9);
  }
  // Ã = W·H, отнимаем сдвиг, который прибавляли
  const Ahat = dot(W, H).map(r => r.map(v => v - shift));
  return { result: { Ahat, W, H }, history };
}
Python
# NMF — мультипликативное обновление
import numpy as np
from sklearn.decomposition import NMF

def nmf_reconstruct(A, k, total_iters=80):
    # Сдвиг для неотрицательности
    shift = -A.min() if A.min() < 0 else 0
    X = A + shift
    # Случайная инициализация
    W = np.random.rand(A.shape[0], k) + 0.5
    H = np.random.rand(k, A.shape[1]) + 0.5
    eps = 1e-9
    for _ in range(total_iters):
        # H ← H · (WᵀX) / (WᵀW·H)
        H *= (W.T @ X) / (W.T @ W @ H + eps)
        # W ← W · (X·Hᵀ) / (W·H·Hᵀ)
        W *= (X @ H.T) / (W @ H @ H.T + eps)
    # Ã = W·H, убираем сдвиг
    Ahat = W @ H - shift
    return Ahat, W, H

# Или через sklearn:
def nmf_sklearn(A, k):
    model = NMF(n_components=k, init='random',
                max_iter=200, random_state=0)
    W = model.fit_transform(A)   # веса тем
    H = model.components_        # сами темы
    Ahat = W @ H
    return Ahat, W, H

4. CUR — разложение по строкам и столбцам

выбор + псевдообратная
Что делает: выбирает k самых «важных» столбцов (C) и k самых «важных» строк (R) по евклидовой норме, строит связующую матрицу U ≈ W⁺ (псевдообратная к пересечению C и R). Итог: A ≈ C·U·R. Где используется: работа с разреженными данными, когда интерпретируемость важнее точности (выбранные столбцы — реальные признаки).
🧩 Пример из жизни: Анализ покупок в интернет-магазине: матрица 10⁶ пользователей × 10⁵ товаров. CUR выбирает 20 реальных товаров (столбцов) и 20 реальных покупателей (строк), давая интерпретируемое приближение — в отличие от SVD, где U и V — абстрактные комбинации.
JavaScript
// CUR — выбор столбцов и строк по норме
export function curReconstruct(A, k) {
  const { m, n } = dims(A);
  const r = Math.max(1, Math.min(k, m, n));

  // Считаем квадраты длин столбцов и строк
  const colNorms = new Array(n).fill(0);
  const rowNorms = new Array(m).fill(0);
  for (let j = 0; j < n; j++)
    for (let i = 0; i < m; i++) colNorms[j] += A[i][j] ** 2;
  for (let i = 0; i < m; i++)
    for (let j = 0; j < n; j++) rowNorms[i] += A[i][j] ** 2;

  // Топ-r столбцов и строк по убыванию нормы
  const topCols = [...Array(n).keys()]
    .sort((a, b) => colNorms[b] - colNorms[a]).slice(0, r);
  const topRows = [...Array(m).keys()]
    .sort((a, b) => rowNorms[b] - rowNorms[a]).slice(0, r);

  // C: выбранные столбцы (m×r), R: выбранные строки (r×n)
  const C = zeros(m, r), R = zeros(r, n);
  const Wcore = zeros(r, r);  // пересечение C ∩ R
  for (let i = 0; i < m; i++)
    for (let t = 0; t < r; t++) C[i][t] = A[i][topCols[t]];
  for (let t = 0; t < r; t++)
    for (let j = 0; j < n; j++) R[t][j] = A[topRows[t]][j];
  for (let i = 0; i < r; i++)
    for (let j = 0; j < r; j++) Wcore[i][j] = A[topRows[i]][topCols[j]];

  // U = W⁺ — псевдообратная через SVD малой матрицы r×r
  const Uc = pinv(Wcore);

  // Ã = C · U · R
  const Ahat = dot(dot(C, Uc), R);
  return { Ahat, C, U: Uc, R, Wcore, topRows, topCols, r };
}
Python
# CUR — выбор по норме + псевдообратная
import numpy as np

def cur_reconstruct(A, k):
    m, n = A.shape
    r = max(1, min(k, m, n))

    # Квадраты длин столбцов и строк
    col_norms = np.linalg.norm(A, axis=0) ** 2
    row_norms = np.linalg.norm(A, axis=1) ** 2

    # Индексы топ-r столбцов и строк
    top_cols = np.argsort(col_norms)[-r:][::-1]
    top_rows = np.argsort(row_norms)[-r:][::-1]

    # C, R и W (пересечение)
    C = A[:, top_cols]          # m×r
    R = A[top_rows, :]          # r×n
    Wcore = A[top_rows[:, None], top_cols]  # r×r

    # U = псевдообратная к W (через SVD)
    Uc = np.linalg.pinv(Wcore)     # r×r

    # Ã = C · U · R
    Ahat = C @ Uc @ R
    return {'Ahat': Ahat, 'C': C,
            'U': Uc, 'R': R,
            'top_rows': top_rows,
            'top_cols': top_cols}

5. ALS — попеременные наименьшие квадраты

итеративное уточнение
Что делает: ищет A ≈ X·Yᵀ, попеременно фиксируя одну матрицу и уточняя другую через МНК с регуляризацией. X ← argmin ||A − X·Yᵀ||² при фиксированном Y (решается через нормальные уравнения). Где используется: рекомендательные системы (Netflix Prize), коллаборативная фильтрация, заполнение пропусков в данных.
🧩 Пример из жизни: Netflix: матрица 500К пользователей × 18К фильмов с рейтингами (много пропусков). ALS находит 50-мерные векторы пользователей (X) и фильмов (Y), предсказывая рейтинг любого фильма для любого пользователя как скалярное произведение xᵤ·yᵢ — основа рекомендаций.
В отличие от NMF, ALS работает с любыми числами (отрицательные тоже), и использует явную регуляризацию λ для борьбы с переобучением.
JavaScript
// ALS — попеременные наименьшие квадраты
export function alsReconstructHistory(A, k, totalIters, reg = 1e-2) {
  const { m, n } = dims(A);
  const r = Math.max(1, Math.min(k, m, n));

  // Случайные начальные X (m×r) и Y (n×r)
  let X = zeros(m, r), Y = zeros(n, r);
  for (let i = 0; i < m; i++)
    for (let t = 0; t < r; t++) X[i][t] = rnd();
  for (let j = 0; j < n; j++)
    for (let t = 0; t < r; t++) Y[j][t] = rnd();

  for (let it = 0; it < totalIters; it++) {
    // --- Фиксируем Y, уточняем X ---
    // Yᵀ·Y + λ·I (r×r — маленькая, решаем быстро)
    const YTY = dot(transpose(Y), Y);
    for (let i = 0; i < r; i++) YTY[i][i] += reg;
    // Для каждой строки i: xᵢ = (YᵀY)⁻¹ · Yᵀ·aᵢ
    for (let i = 0; i < m; i++) {
      const rhs = new Array(r).fill(0);
      for (let t = 0; t < r; t++)
        for (let j = 0; j < n; j++) rhs[t] += Y[j][t] * A[i][j];
      const sol = numeric.solve(YTY, rhs);
      for (let t = 0; t < r; t++) X[i][t] = sol[t];
    }
    // --- Фиксируем X, уточняем Y ---
    const XTX = dot(transpose(X), X);
    for (let i = 0; i < r; i++) XTX[i][i] += reg;
    for (let j = 0; j < n; j++) {
      const rhs = new Array(r).fill(0);
      for (let t = 0; t < r; t++)
        for (let i = 0; i < m; i++) rhs[t] += X[i][t] * A[i][j];
      const sol = numeric.solve(XTX, rhs);
      for (let t = 0; t < r; t++) Y[j][t] = sol[t];
    }
  }
  // Ã = X·Yᵀ
  const Ahat = dot(X, transpose(Y));
  return { result: { Ahat, X, Y, r, reg }, history };
}
Python
# ALS — попеременные наименьшие квадраты
import numpy as np

def als_reconstruct(A, k, total_iters=80, reg=1e-2):
    m, n = A.shape
    r = min(k, m, n)

    # Случайная инициализация
    X = np.random.randn(m, r) * 0.5
    Y = np.random.randn(n, r) * 0.5

    I = np.eye(r)  # единичная для регуляризации

    for _ in range(total_iters):
        # --- Шаг X: X = A·Y · (YᵀY + λ·I)⁻¹ ---
        YTY = Y.T @ Y + reg * I          # r×r
        X = (A @ Y) @ np.linalg.inv(YTY)  # m×r

        # --- Шаг Y: Y = Aᵀ·X · (XᵀX + λ·I)⁻¹ ---
        XTX = X.T @ X + reg * I          # r×r
        Y = (A.T @ X) @ np.linalg.inv(XTX)  # n×r

    # Ã = X·Yᵀ
    Ahat = X @ Y.T
    return Ahat, X, Y

# Компактная версия через solve (численно устойчивее):
def als_solve(A, k, total_iters=80, reg=1e-2):
    m, n = A.shape
    r = min(k, m, n)
    X = np.random.randn(m, r) * 0.5
    Y = np.random.randn(n, r) * 0.5
    I = np.eye(r)
    for _ in range(total_iters):
        X = np.linalg.solve(Y.T @ Y + reg * I, (A @ Y).T).T
        Y = np.linalg.solve(X.T @ X + reg * I, (A.T @ X).T).T
    return X @ Y.T, X, Y
